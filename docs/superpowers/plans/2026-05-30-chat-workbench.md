# Chat Workbench Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first vertical slice of the Chat-driven workspace: Chat entry, `Auto/Ask/Agent` mode switching, Script/Workflow Skills, and async task status flow.

**Architecture:** Add a new workbench layer above existing Coze capabilities instead of rewriting Agent IDE, Workflow IDE, or resource library. Backend additions follow the repo's IDL generated handler plus application/domain pattern; frontend additions extend the existing space route and submenu while reusing current `develop` and `library` pages.

**Tech Stack:** Go, Hertz, Thrift IDL, GORM/MySQL, existing coderunner, existing workflow service, React 18, TypeScript, Rsbuild, Semi/Coze Design, Vitest.

---

## Scope Check

The approved spec touches frontend navigation, Skill Registry, Task Engine, and Chat Orchestrator. This plan keeps them as one vertical-slice project because each task builds toward one user-visible flow: submit a Chat request, route it to Ask/Agent/Skill/Task, and see the result or task progress. Execute tasks in order; each task has its own verification and commit point.

```go
package service

import (
	"encoding/json"
	"fmt"
	"strings"

	"gopkg.in/yaml.v3"
)

type Declaration struct {
	ID           string                 `json:"id" yaml:"id"`
	Name         string                 `json:"name" yaml:"name"`
	Description  string                 `json:"description" yaml:"description"`
	Type         string                 `json:"type" yaml:"type"`
	Version      string                 `json:"version" yaml:"version"`
	Enabled      bool                   `json:"enabled" yaml:"enabled"`
	InputSchema  map[string]any         `json:"input_schema" yaml:"input_schema"`
	OutputSchema map[string]any         `json:"output_schema" yaml:"output_schema"`
	Executor     ExecutorDeclaration    `json:"executor" yaml:"executor"`
	Permissions  PermissionDeclaration  `json:"permissions" yaml:"permissions"`
}

type ExecutorDeclaration struct {
	Language   string `json:"language,omitempty" yaml:"language,omitempty"`
	Entry      string `json:"entry,omitempty" yaml:"entry,omitempty"`
	WorkflowID string `json:"workflow_id,omitempty" yaml:"workflow_id,omitempty"`
	Version    string `json:"version,omitempty" yaml:"version,omitempty"`
}

type PermissionDeclaration struct {
	Network         bool     `json:"network" yaml:"network"`
	FilesystemRead  []string `json:"filesystem_read,omitempty" yaml:"filesystem_read,omitempty"`
	FilesystemWrite []string `json:"filesystem_write,omitempty" yaml:"filesystem_write,omitempty"`
}

func ParseDeclaration(fileName string, content []byte) (*Declaration, error) {
	var decl Declaration
	name := strings.ToLower(fileName)
	switch {
	case strings.HasSuffix(name, ".yaml"), strings.HasSuffix(name, ".yml"):
		if err := yaml.Unmarshal(content, &decl); err != nil {
			return nil, fmt.Errorf("parse yaml declaration: %w", err)
		}
	case strings.HasSuffix(name, ".json"):
		if err := json.Unmarshal(content, &decl); err != nil {
			return nil, fmt.Errorf("parse json declaration: %w", err)
		}
	default:
		return nil, fmt.Errorf("unsupported declaration file: %s", fileName)
	}
	if err := ValidateDeclaration(&decl); err != nil {
		return nil, err
	}
	return &decl, nil
}

func ValidateDeclaration(decl *Declaration) error {
	if decl == nil {
		return fmt.Errorf("declaration is required")
	}
	if strings.TrimSpace(decl.ID) == "" {
		return fmt.Errorf("id is required")
	}
	if strings.TrimSpace(decl.Name) == "" {
		return fmt.Errorf("name is required")
	}
	if strings.TrimSpace(decl.Type) == "" {
		return fmt.Errorf("type is required")
	}
	switch decl.Type {
	case "script":
		if decl.Executor.Language != "python" {
			return fmt.Errorf("script executor language must be python")
		}
		if strings.TrimSpace(decl.Executor.Entry) == "" {
			return fmt.Errorf("script executor entry is required")
		}
	case "workflow":
		if strings.TrimSpace(decl.Executor.WorkflowID) == "" {
			return fmt.Errorf("workflow_id is required")
		}
	default:
		return fmt.Errorf("unsupported skill type: %s", decl.Type)
	}
	return nil
}
```

```go
		entity.StatusFailed:   true,
	},
	entity.StatusFailed: {
		entity.StatusQueued: true,
	},
}

func CanTransition(from, to entity.Status) bool {
	return allowedTransitions[from][to]
}

func EnsureTransition(from, to entity.Status) error {
	if CanTransition(from, to) {
		return nil
```

```go
	Code   string

func (e *ScriptExecutor) Run(ctx context.Context, skill *Declaration, input map[string]any) (map[string]any, error) {
	if e.Runner == nil {

	}
	return resp.Result, nil
}
```

```go
package workbench

import "strings"

type IntentKind string

const (
	IntentChat  IntentKind = "chat"
	IntentAgent IntentKind = "agent"
	IntentSkill IntentKind = "skill"
	IntentTask  IntentKind = "task"
)

type Intent struct {
	Kind          IntentKind
	Confidence    float64
	MatchedSkillID int64
	RequiresAsync bool
	Reason        string
}

func ResolveIntent(message string, selectedSkillID int64) Intent {
	lower := strings.ToLower(message)
	if selectedSkillID > 0 {
		return Intent{Kind: IntentSkill, Confidence: 1, MatchedSkillID: selectedSkillID, Reason: "selected skill"}
	}
	if strings.Contains(lower, "报告") || strings.Contains(lower, "分析") || strings.Contains(lower, "整理") {
		return Intent{Kind: IntentSkill, Confidence: 0.75, RequiresAsync: true, Reason: "long running task keyword"}
	}
	return Intent{Kind: IntentChat, Confidence: 0.7, Reason: "default chat"}
}
```

Create `backend/application/workbench/router_decision.go`:

```go
package workbench

type ChatMode string

const (
	ChatModeAuto  ChatMode = "auto"
	ChatModeAsk   ChatMode = "ask"
	ChatModeAgent ChatMode = "agent"
)

type RouteTarget string

const (
	RouteChatDirect  RouteTarget = "chat_direct"
	RouteAgentEngine RouteTarget = "agent_engine"
	RouteSkillEngine RouteTarget = "skill_engine"
	RouteTaskEngine  RouteTarget = "task_engine"
)

type RouteDecision struct {
	Target RouteTarget
	Reason string
}

func DecideRoute(mode ChatMode, intent Intent) RouteDecision {
	switch mode {
	case ChatModeAsk:
		return RouteDecision{Target: RouteChatDirect, Reason: "ask mode"}
	case ChatModeAgent:
		return RouteDecision{Target: RouteAgentEngine, Reason: "agent mode"}
	default:
		if intent.RequiresAsync || intent.Kind == IntentTask {
			return RouteDecision{Target: RouteTaskEngine, Reason: intent.Reason}
		}
		if intent.Kind == IntentSkill {
			return RouteDecision{Target: RouteSkillEngine, Reason: intent.Reason}
		}
		if intent.Kind == IntentAgent {
			return RouteDecision{Target: RouteAgentEngine, Reason: intent.Reason}
		}
		return RouteDecision{Target: RouteChatDirect, Reason: intent.Reason}
	}
}
```
